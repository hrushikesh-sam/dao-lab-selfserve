-- effective schema of a fresh origin/rama (747d10d) DB, user_version 11
CREATE TABLE users (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    username    TEXT    NOT NULL UNIQUE,
    role        TEXT    NOT NULL DEFAULT 'user'
                CHECK (role IN ('admin', 'user')),
    enabled     INTEGER NOT NULL DEFAULT 1 CHECK (enabled IN (0, 1)),
    token_hash  TEXT    NOT NULL UNIQUE,          -- SHA-256 hex of the bearer token
    ssh_pubkey  TEXT    NOT NULL,
    created_at  TEXT    NOT NULL,                 -- UTC ISO-8601 YYYY-MM-DDTHH:MM:SSZ
    updated_at  TEXT    NOT NULL
, token_expires_at TEXT);

CREATE TABLE setups (
    id           INTEGER PRIMARY KEY AUTOINCREMENT,
    name         TEXT    NOT NULL UNIQUE,         -- e.g. 'setup-01'
    pool_id      INTEGER NOT NULL REFERENCES pools(id),
    capabilities TEXT    NOT NULL DEFAULT '[]',   -- JSON array; computed union, sorted
    links        TEXT    NOT NULL DEFAULT '[]',   -- JSON array of {a,b,capability}
    diagram      TEXT    NOT NULL DEFAULT '',     -- mermaid source, rendered verbatim
    yaml_path    TEXT    NOT NULL,                -- e.g. 'setups/setup-01.yml'
    created_at   TEXT    NOT NULL,
    updated_at   TEXT    NOT NULL
, availability TEXT NOT NULL DEFAULT 'available' CHECK (availability IN ('available', 'quarantined', 'maintenance')));

CREATE TABLE reservations (
    id                 INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id            INTEGER NOT NULL REFERENCES users(id),
    usecase_id         INTEGER NOT NULL REFERENCES usecases(id),
    setup_id           INTEGER NOT NULL REFERENCES setups(id),
    state              TEXT    NOT NULL DEFAULT 'queued'
                       CHECK (state IN ('queued', 'provisioning', 'ready', 'active',
                                        'cleaning', 'quarantined', 'released', 'failed')),
    requested_start_at TEXT    NOT NULL,          -- what the user asked for
    start_at           TEXT    NOT NULL,          -- what the scheduler granted (slot-shift)
    end_at             TEXT    NOT NULL,
    diy                INTEGER NOT NULL DEFAULT 0 CHECK (diy IN (0, 1)),
    diy_templates      TEXT,                      -- JSON object host -> template; NULL unless diy
    retries            INTEGER NOT NULL DEFAULT 0,
    failure_reason     TEXT,                      -- set only in state 'failed'
    created_at         TEXT    NOT NULL,
    updated_at         TEXT    NOT NULL
, mode TEXT NOT NULL DEFAULT 'asap' CHECK (mode IN ('asap', 'at')), turnaround_min INTEGER NOT NULL DEFAULT 0, ssh_pubkey TEXT, access_password TEXT);
CREATE INDEX idx_reservations_setup    ON reservations(setup_id, start_at, end_at);
CREATE INDEX idx_reservations_state    ON reservations(state);
CREATE INDEX idx_reservations_user     ON reservations(user_id);

CREATE TABLE jobs (
            id             INTEGER PRIMARY KEY AUTOINCREMENT,
            reservation_id INTEGER NOT NULL REFERENCES reservations(id) ON DELETE CASCADE,
            kind           TEXT    NOT NULL
                           CHECK (kind IN ('deploy', 'cleanup', 'health_probe', 'reconcile')),
            state          TEXT    NOT NULL DEFAULT 'pending'
                           CHECK (state IN ('pending', 'running', 'succeeded', 'failed')),
            attempt        INTEGER NOT NULL DEFAULT 1 CHECK (attempt >= 1),
            trace_id       TEXT,
            log_path       TEXT,
            steps          TEXT    NOT NULL DEFAULT '[]',   -- JSON list of step records
            error          TEXT,                            -- set only in state 'failed'
            started_at     TEXT,                            -- set when the job starts running
            finished_at    TEXT,                            -- set when it succeeds or fails
            created_at     TEXT    NOT NULL,
            updated_at     TEXT    NOT NULL
        );
CREATE INDEX idx_jobs_reservation ON jobs(reservation_id, id);
CREATE INDEX idx_jobs_state ON jobs(state, id);
